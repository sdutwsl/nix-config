cmake_minimum_required(VERSION 3.16)
project(sogoubridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Fcitx5Core REQUIRED)
find_package(Fcitx5ModuleCloudPinyin REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(LibSystemd REQUIRED IMPORTED_TARGET libsystemd)
add_library(sogoubridge MODULE bridge.cpp)
target_link_libraries(sogoubridge PRIVATE Fcitx5::Core Fcitx5::Module::CloudPinyin PkgConfig::LibSystemd)
set_target_properties(sogoubridge PROPERTIES PREFIX "")
install(TARGETS sogoubridge DESTINATION lib/fcitx5)
install(FILES sogoubridge.conf DESTINATION share/fcitx5/addon)
install(FILES sogou.conf DESTINATION share/fcitx5/inputmethod)
